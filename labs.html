Solution to 3 — the ticks pattern
solution_ticks.py
# Lab 1 - Try-this solution: two LEDs at two rates WITHOUT sleep()
# Red LED on GPIO 26 toggles every 500 ms, green LED on GPIO 27 every 130 ms.
from machine import Pin
import time

red = Pin(26, Pin.OUT)
green = Pin(27, Pin.OUT)
last_red = last_green = time.ticks_ms()     # time of the last toggle

while True:
    now = time.ticks_ms()                    # ms since the board started
    if time.ticks_diff(now, last_red) >= 500:
        last_red = now
        red.value(not red.value())
    if time.ticks_diff(now, last_green) >= 130:
        last_green = now
        green.value(not green.value())
    # the loop never blocks, so the CPU is free for other work here

Big idea: instead of waiting, remember when you last acted and check the clock with time.ticks_ms(). The loop stays free — this is cooperative multitasking. (In Arduino C the same function is called millis().)

solution_millis.ino
// Lab 1 - Try-this solution: two LEDs at two different rates WITHOUT delay()
// Red LED on GPIO 26 toggles every 500 ms, green LED on GPIO 27 every 130 ms.

const int RED_PIN = 26;
const int GREEN_PIN = 27;

unsigned long lastRed = 0, lastGreen = 0;   // time of the last toggle
bool redOn = false, greenOn = false;

void setup() {
  pinMode(RED_PIN, OUTPUT);
  pinMode(GREEN_PIN, OUTPUT);
}

void loop() {
  unsigned long now = millis();             // ms since the board started

  if (now - lastRed >= 500) {               // is it time to toggle red?
    lastRed = now;
    redOn = !redOn;
    digitalWrite(RED_PIN, redOn);
  }
  if (now - lastGreen >= 130) {             // is it time to toggle green?
    lastGreen = now;
    greenOn = !greenOn;
    digitalWrite(GREEN_PIN, greenOn);
  }
  // loop() never blocks, so the CPU is free to do other work here
}

# Lab 2 - Buttons & Interrupts
# Push-button on GPIO 4 (other leg to GND), LED on GPIO 26.
# Every press toggles the LED and increments a counter - handled by an INTERRUPT.
from machine import Pin
import time

led = Pin(26, Pin.OUT)
button = Pin(4, Pin.IN, Pin.PULL_UP)   # internal pull-up: idle = 1, pressed = 0

pressed = False        # set inside the handler, read in the main loop
last_irq = 0           # for debouncing

def on_button(pin):
    # Interrupt handler: keep it SHORT. No print(), no sleep() here.
    global pressed, last_irq
    now = time.ticks_ms()
    if time.ticks_diff(now, last_irq) > 200:   # ignore contact bounce
        pressed = True
        last_irq = now

button.irq(trigger=Pin.IRQ_FALLING, handler=on_button)
print("Lab 2: press the button!")

count = 0
while True:
    if pressed:                 # the handler raised the flag
        pressed = False
        count += 1
        led.value(not led.value())
        print("Press #%d -> LED %s" % (count, "ON" if led.value() else "OFF"))
    time.sleep_ms(10)           # the CPU is free here for other work

Try this

  1. Change the debounce window from 200 ms to 0. Click quickly — do you ever see a press counted twice? That is contact bounce.
  2. Why does on_button() need the line global pressed, last_irq?
  3. Add time.sleep(3) inside the main loop in both versions. Press the button during the sleep. Which version still notices the press?
  4. Count only long presses (held > 1 s). Deep dive
Answers
  1. Real switches “chatter” for a few milliseconds, creating several edges. The time window ignores them.
  2. Without global, the handler would create new local variables and the main loop would never see the flag change. (In C the same idea needs the volatile keyword.)
  3. The interrupt version: the handler sets the flag even while the main loop sleeps, and the press is handled when it wakes. The polling version misses it.
  4. Use Pin.IRQ_FALLING | Pin.IRQ_RISING, store the press time on 0 and compare on 1; or poll and time with ticks_ms().

// Lab 2 - Buttons & Interrupts
// Push-button on GPIO 4 (other leg to GND), LED on GPIO 26.
// Every press toggles the LED and increments a counter - handled by an INTERRUPT.

const int BUTTON_PIN = 4;
const int LED_PIN = 26;

volatile bool pressed = false;        // set inside the ISR, read in loop()
volatile unsigned long lastIsrMs = 0; // for debouncing

// Interrupt Service Routine: keep it SHORT. No Serial.print, no delay here.
void IRAM_ATTR onButton() {
  unsigned long now = millis();
  if (now - lastIsrMs > 200) {        // ignore contact bounce (< 200 ms)
    pressed = true;
    lastIsrMs = now;
  }
}

int count = 0;
bool ledOn = false;

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);  // internal pull-up: idle = HIGH, pressed = LOW
  attachInterrupt(digitalPinToInterrupt(BUTTON_PIN), onButton, FALLING);
  Serial.println("Lab 2: press the button!");
}

void loop() {
  if (pressed) {                      // the ISR raised the flag
    pressed = false;
    count++;
    ledOn = !ledOn;
    digitalWrite(LED_PIN, ledOn);
    Serial.printf("Press #%d -> LED %s\n", count, ledOn ? "ON" : "OFF");
  }
  // The CPU is free here - e.g. blink something else, read sensors...
}

# Lab 3 - The Analog World: ADC in, PWM out
# Potentiometer wiper -> GPIO 34 (ADC), LED on GPIO 26 dimmed with PWM.
from machine import Pin, ADC, PWM
import time

pot = ADC(Pin(34))
pot.atten(ADC.ATTN_11DB)             # full 0 ... 3.3 V input range
led = PWM(Pin(26), freq=1000)        # 1 kHz PWM on the LED

while True:
    raw = pot.read_u16() >> 4        # 16-bit reading -> 12-bit: 0 ... 4095
    volts = raw * 3.3 / 4095
    duty = raw * 255 // 4095         # scale to 8-bit duty: 0 ... 255
    led.duty_u16(duty * 257)         # 0..255 -> 0..65535
    print("raw:%d volts:%.2f duty:%d" % (raw, volts, duty))
    time.sleep_ms(100)
Click the potentiometer in the simulation and drag its knob. read_u16() gives 0–65535; shifting right by 4 bits turns it into the ESP32’s native 12-bit range, 0–4095.

🧮 ECE calculations

  1. Quantization: one ADC step (LSB) = 3.3 V / 4096. The quantization error is at most ±½ LSB. What is the ideal signal-to-quantization-noise ratio of a 12-bit ADC? (Use SQNR ≈ 6.02 N + 1.76 dB.)
  2. PWM to true analog: an RC low-pass filter turns PWM into a steady voltage. With R = 10 kΩ and C = 1 µF, what is the cut-off frequency fc = 1/(2πRC)? Is it well below the 1 kHz PWM frequency?
  3. Change freq=1000 to freq=20. What do you see on the LED, and why does PWM frequency matter for lamps and motors?
Answers
  1. LSB ≈ 0.81 mV; SQNR ≈ 6.02 × 12 + 1.76 ≈ 74 dB (the real ESP32 ADC is worse, because of noise and non-linearity).
  2. fc ≈ 16 Hz, far below 1 kHz, so the ripple is small and the output is the average: duty × 3.3 V.
  3. At 20 Hz the LED visibly flickers. Lamps need well above about 100 Hz; motor drivers often use 20 kHz or more so the switching is not audible.

# MicroPython SSD1306 OLED driver, I2C and SPI interfaces

from micropython import const
import framebuf


# register definitions
SET_CONTRAST = const(0x81)
SET_ENTIRE_ON = const(0xA4)
SET_NORM_INV = const(0xA6)
SET_DISP = const(0xAE)
SET_MEM_ADDR = const(0x20)
SET_COL_ADDR = const(0x21)
SET_PAGE_ADDR = const(0x22)
SET_DISP_START_LINE = const(0x40)
SET_SEG_REMAP = const(0xA0)
SET_MUX_RATIO = const(0xA8)
SET_IREF_SELECT = const(0xAD)
SET_COM_OUT_DIR = const(0xC0)
SET_DISP_OFFSET = const(0xD3)
SET_COM_PIN_CFG = const(0xDA)
SET_DISP_CLK_DIV = const(0xD5)
SET_PRECHARGE = const(0xD9)
SET_VCOM_DESEL = const(0xDB)
SET_CHARGE_PUMP = const(0x8D)


# Subclassing FrameBuffer provides support for graphics primitives
# http://docs.micropython.org/en/latest/pyboard/library/framebuf.html
class SSD1306(framebuf.FrameBuffer):
    def __init__(self, width, height, external_vcc):
        self.width = width
        self.height = height
        self.external_vcc = external_vcc
        self.pages = self.height // 8
        self.buffer = bytearray(self.pages * self.width)
        super().__init__(self.buffer, self.width, self.height, framebuf.MONO_VLSB)
        self.init_display()

    def init_display(self):
        for cmd in (
            SET_DISP,  # display off
            # address setting
            SET_MEM_ADDR,
            0x00,  # horizontal
            # resolution and layout
            SET_DISP_START_LINE,  # start at line 0
            SET_SEG_REMAP | 0x01,  # column addr 127 mapped to SEG0
            SET_MUX_RATIO,
            self.height - 1,
            SET_COM_OUT_DIR | 0x08,  # scan from COM[N] to COM0
            SET_DISP_OFFSET,
            0x00,
            SET_COM_PIN_CFG,
            0x02 if self.width > 2 * self.height else 0x12,
            # timing and driving scheme
            SET_DISP_CLK_DIV,
            0x80,
            SET_PRECHARGE,
            0x22 if self.external_vcc else 0xF1,
            SET_VCOM_DESEL,
            0x30,  # 0.83*Vcc
            # display
            SET_CONTRAST,
            0xFF,  # maximum
            SET_ENTIRE_ON,  # output follows RAM contents
            SET_NORM_INV,  # not inverted
            SET_IREF_SELECT,
            0x30,  # enable internal IREF during display on
            # charge pump
            SET_CHARGE_PUMP,
            0x10 if self.external_vcc else 0x14,
            SET_DISP | 0x01,  # display on
        ):  # on
            self.write_cmd(cmd)
        self.fill(0)
        self.show()

    def poweroff(self):
        self.write_cmd(SET_DISP)

    def poweron(self):
        self.write_cmd(SET_DISP | 0x01)

    def contrast(self, contrast):
        self.write_cmd(SET_CONTRAST)
        self.write_cmd(contrast)

    def invert(self, invert):
        self.write_cmd(SET_NORM_INV | (invert & 1))

    def rotate(self, rotate):
        self.write_cmd(SET_COM_OUT_DIR | ((rotate & 1) << 3))
        self.write_cmd(SET_SEG_REMAP | (rotate & 1))

    def show(self):
        x0 = 0
        x1 = self.width - 1
        if self.width != 128:
            # narrow displays use centred columns
            col_offset = (128 - self.width) // 2
            x0 += col_offset
            x1 += col_offset
        self.write_cmd(SET_COL_ADDR)
        self.write_cmd(x0)
        self.write_cmd(x1)
        self.write_cmd(SET_PAGE_ADDR)
        self.write_cmd(0)
        self.write_cmd(self.pages - 1)
        self.write_data(self.buffer)


class SSD1306_I2C(SSD1306):
    def __init__(self, width, height, i2c, addr=0x3C, external_vcc=False):
        self.i2c = i2c
        self.addr = addr
        self.temp = bytearray(2)
        self.write_list = [b"\x40", None]  # Co=0, D/C#=1
        super().__init__(width, height, external_vcc)

    def write_cmd(self, cmd):
        self.temp[0] = 0x80  # Co=1, D/C#=0
        self.temp[1] = cmd
        self.i2c.writeto(self.addr, self.temp)

    def write_data(self, buf):
        self.write_list[1] = buf
        self.i2c.writevto(self.addr, self.write_list)


class SSD1306_SPI(SSD1306):
    def __init__(self, width, height, spi, dc, res, cs, external_vcc=False):
        self.rate = 10 * 1024 * 1024
        dc.init(dc.OUT, value=0)
        res.init(res.OUT, value=0)
        cs.init(cs.OUT, value=1)
        self.spi = spi
        self.dc = dc
        self.res = res
        self.cs = cs
        import time

        self.res(1)
        time.sleep_ms(1)
        self.res(0)
        time.sleep_ms(10)
        self.res(1)
        super().__init__(width, height, external_vcc)

    def write_cmd(self, cmd):
        self.spi.init(baudrate=self.rate, polarity=0, phase=0)
        self.cs(1)
        self.dc(0)
        self.cs(0)
        self.spi.write(bytearray([cmd]))
        self.cs(1)

    def write_data(self, buf):
        self.spi.init(baudrate=self.rate, polarity=0, phase=0)
        self.cs(1)
        self.dc(1)
        self.cs(0)
        self.spi.write(buf)
        self.cs(1)

# Lab 5 - Your ESP32 on the Internet (MQTT)
# DHT22 -> GPIO 15, LED -> GPIO 26. Joins Wokwi's virtual Wi-Fi, publishes
# readings every 5 s, and listens for ON/OFF commands from the class dashboard.
import network
import time
import dht
import machine
from machine import Pin
from umqtt.simple import MQTTClient

# ======= CHANGE THIS: your name or USN (letters/numbers only, no spaces) =======
STUDENT = "your-name"
# ===============================================================================

# Free public brokers, tried in order (the class dashboard listens to all of them)
BROKERS = ["broker.hivemq.com", "broker.emqx.io", "test.mosquitto.org"]
TOPIC_ROOT = "navkis/ews2026"          # must match the dashboard

sensor = dht.DHT22(Pin(15))
led = Pin(26, Pin.OUT)
data_topic = "%s/%s/data" % (TOPIC_ROOT, STUDENT)
led_topic = "%s/%s/led" % (TOPIC_ROOT, STUDENT)

print("Connecting to Wi-Fi", end="")
wlan = network.WLAN(network.STA_IF)
wlan.active(True)
wlan.connect("Wokwi-GUEST", "")        # open network
while not wlan.isconnected():
    print(".", end="")
    time.sleep(0.1)
print(" connected! IP =", wlan.ifconfig()[0])

def on_message(topic, msg):            # runs when a subscribed message arrives
    print("Message on", topic.decode(), ":", msg.decode())
    if msg == b"ON":
        led.value(1)
    elif msg == b"OFF":
        led.value(0)

client_id = "navkis-%s-%d" % (STUDENT, machine.unique_id()[-1])
client = None
for broker in BROKERS:                 # use the first broker that answers
    try:
        print("Connecting to MQTT broker", broker, "...", end="")
        c = MQTTClient(client_id, broker, keepalive=60)
        c.set_callback(on_message)
        c.connect()
        client = c
        break
    except Exception as e:
        print(" failed:", e)
if client is None:
    raise SystemExit("No MQTT broker reachable - tell the instructor")
client.subscribe(led_topic)
print(" connected! Listening on", led_topic)

last_send = time.ticks_ms() - 5000
while True:
    client.check_msg()                 # handle incoming messages (non-blocking)
    if time.ticks_diff(time.ticks_ms(), last_send) >= 5000:   # every 5 seconds
        last_send = time.ticks_ms()
        try:
            sensor.measure()
            msg = '{"t":%.1f,"h":%.1f,"led":%d}' % (sensor.temperature(), sensor.humidity(), led.value())
            client.publish(data_topic, msg)
            print("Published to", data_topic, ":", msg)
        except OSError as e:
            print("Sensor error:", e)
    time.sleep_ms(50)

What is happening?

  1. Wokwi provides a virtual access point, Wokwi-GUEST (no password), connected to the real Internet.
  2. Your ESP32 opens a TCP connection to the public MQTT broker on port 1883 (umqtt.simple is built into MicroPython).
  3. Every 5 s it publishes a small JSON message to navkis/ews2026/<your-name>/data.
  4. It subscribes to navkis/ews2026/<your-name>/led. When anyone publishes ON or OFF there, your LED obeys.
  5. The Live Dashboard subscribes to navkis/ews2026/# and shows everyone at once.
A public broker is like a public notice board: anyone can read your topic. Never send passwords or personal data this way. Real products use TLS and a username and password.

# Lab 6 (Deep dive) - Cooperative multitasking with asyncio
# Three independent tasks share one CPU, like tasks in a real-time OS:
#   blink_task  - toggles the LED on GPIO 26 every 250 ms
#   sensor_task - reads the potentiometer (GPIO 34) and puts values in a FIFO
#   print_task  - waits until data arrives, then prints it
# Potentiometer wiper -> GPIO 34, LED -> GPIO 26 (same wiring as Lab 3).
import asyncio
from machine import Pin, ADC

led = Pin(26, Pin.OUT)
pot = ADC(Pin(34))
pot.atten(ADC.ATTN_11DB)

fifo = []                      # a simple queue between producer and consumer
data_ready = asyncio.Event()   # lets print_task sleep until there is data

async def blink_task():
    while True:
        led.value(not led.value())
        await asyncio.sleep_ms(250)    # give the CPU to the other tasks

async def sensor_task():               # producer
    while True:
        if len(fifo) < 10:             # drop the sample if the queue is full
            fifo.append(pot.read_u16() >> 4)
            data_ready.set()
        await asyncio.sleep_ms(500)

async def print_task():                # consumer
    while True:
        await data_ready.wait()        # block until the producer signals
        data_ready.clear()
        while fifo:
            print("pot =", fifo.pop(0))

async def main():
    asyncio.create_task(blink_task())
    asyncio.create_task(sensor_task())
    asyncio.create_task(print_task())
    while True:                        # main() is just another task
        await asyncio.sleep(1)

asyncio.run(main())
asyncio vs FreeRTOS. Here tasks cooperate: each one gives up the CPU at every await. In FreeRTOS (the C version below) the scheduler can preempt a task at any time by priority. Same ideas — tasks, blocking, producer–consumer — with a different scheduling policy.

// Lab 6 (Deep dive) - Real-time multitasking with FreeRTOS
// Three independent tasks share one CPU, like tasks in a real-time OS:
//   blinkTask  - toggles the LED on GPIO 26 every 250 ms
//   sensorTask - reads the potentiometer (GPIO 34) and sends values to a QUEUE
//   printTask  - waits on the queue and prints whatever arrives
// Potentiometer wiper -> GPIO 34, LED -> GPIO 26 (same wiring as Lab 3).

const int LED_PIN = 26;
const int POT_PIN = 34;

QueueHandle_t potQueue;               // FIFO between sensorTask and printTask

void blinkTask(void* param) {
  pinMode(LED_PIN, OUTPUT);
  while (true) {
    digitalWrite(LED_PIN, !digitalRead(LED_PIN));
    vTaskDelay(pdMS_TO_TICKS(250));   // sleep this task; others keep running
  }
}

void sensorTask(void* param) {
  while (true) {
    int value = analogRead(POT_PIN);
    xQueueSend(potQueue, &value, 0);  // drop the sample if the queue is full
    vTaskDelay(pdMS_TO_TICKS(500));
  }
}

void printTask(void* param) {
  int value;
  while (true) {
    if (xQueueReceive(potQueue, &value, portMAX_DELAY)) {  // block until data
      Serial.printf("[core %d] pot = %d\n", xPortGetCoreID(), value);
    }
  }
}

void setup() {
  Serial.begin(115200);
  potQueue = xQueueCreate(10, sizeof(int));   // holds up to 10 ints

  //                       function    name      stack  param prio handle
  xTaskCreate(blinkTask,  "blink",   2048,  NULL, 1,   NULL);
  xTaskCreate(sensorTask, "sensor",  2048,  NULL, 2,   NULL);
  xTaskCreate(printTask,  "print",   4096,  NULL, 1,   NULL);
}

void loop() {
  vTaskDelay(pdMS_TO_TICKS(1000));    // loop() is itself just another task
}

# Normal variables live in RAM, which is switched off in deep sleep.

rtc.memory(str(boots))

if machine.reset_cause() == machine.DEEPSLEEP_RESET:

from micropython import const
import framebuf


# register definitions
SET_CONTRAST = const(0x81)
SET_ENTIRE_ON = const(0xA4)
SET_NORM_INV = const(0xA6)
SET_DISP = const(0xAE)
SET_MEM_ADDR = const(0x20)
SET_COL_ADDR = const(0x21)
SET_PAGE_ADDR = const(0x22)
SET_DISP_START_LINE = const(0x40)
SET_SEG_REMAP = const(0xA0)
SET_MUX_RATIO = const(0xA8)
SET_IREF_SELECT = const(0xAD)
SET_COM_OUT_DIR = const(0xC0)
SET_DISP_OFFSET = const(0xD3)
SET_COM_PIN_CFG = const(0xDA)
SET_DISP_CLK_DIV = const(0xD5)
SET_PRECHARGE = const(0xD9)
SET_VCOM_DESEL = const(0xDB)
SET_CHARGE_PUMP = const(0x8D)


# Subclassing FrameBuffer provides support for graphics primitives
# http://docs.micropython.org/en/latest/pyboard/library/framebuf.html
class SSD1306(framebuf.FrameBuffer):
    def __init__(self, width, height, external_vcc):
        self.width = width
        self.height = height
        self.external_vcc = external_vcc
        self.pages = self.height // 8
        self.buffer = bytearray(self.pages * self.width)
        super().__init__(self.buffer, self.width, self.height, framebuf.MONO_VLSB)
        self.init_display()

    def init_display(self):
        for cmd in (
            SET_DISP,  # display off
            # address setting
            SET_MEM_ADDR,
            0x00,  # horizontal
            # resolution and layout
            SET_DISP_START_LINE,  # start at line 0
            SET_SEG_REMAP | 0x01,  # column addr 127 mapped to SEG0
            SET_MUX_RATIO,
            self.height - 1,
            SET_COM_OUT_DIR | 0x08,  # scan from COM[N] to COM0
            SET_DISP_OFFSET,
            0x00,
            SET_COM_PIN_CFG,
            0x02 if self.width > 2 * self.height else 0x12,
            # timing and driving scheme
            SET_DISP_CLK_DIV,
            0x80,
            SET_PRECHARGE,
            0x22 if self.external_vcc else 0xF1,
            SET_VCOM_DESEL,
            0x30,  # 0.83*Vcc
            # display
            SET_CONTRAST,
            0xFF,  # maximum
            SET_ENTIRE_ON,  # output follows RAM contents
            SET_NORM_INV,  # not inverted
            SET_IREF_SELECT,
            0x30,  # enable internal IREF during display on
            # charge pump
            SET_CHARGE_PUMP,
            0x10 if self.external_vcc else 0x14,
            SET_DISP | 0x01,  # display on
        ):  # on
            self.write_cmd(cmd)
        self.fill(0)
        self.show()

    def poweroff(self):
        self.write_cmd(SET_DISP)

    def poweron(self):
        self.write_cmd(SET_DISP | 0x01)

    def contrast(self, contrast):
        self.write_cmd(SET_CONTRAST)
        self.write_cmd(contrast)

    def invert(self, invert):
        self.write_cmd(SET_NORM_INV | (invert & 1))

    def rotate(self, rotate):
        self.write_cmd(SET_COM_OUT_DIR | ((rotate & 1) << 3))
        self.write_cmd(SET_SEG_REMAP | (rotate & 1))

    def show(self):
        x0 = 0
        x1 = self.width - 1
        if self.width != 128:
            # narrow displays use centred columns
            col_offset = (128 - self.width) // 2
            x0 += col_offset
            x1 += col_offset
        self.write_cmd(SET_COL_ADDR)
        self.write_cmd(x0)
        self.write_cmd(x1)
        self.write_cmd(SET_PAGE_ADDR)
        self.write_cmd(0)
        self.write_cmd(self.pages - 1)
        self.write_data(self.buffer)


class SSD1306_I2C(SSD1306):
    def __init__(self, width, height, i2c, addr=0x3C, external_vcc=False):
        self.i2c = i2c
        self.addr = addr
        self.temp = bytearray(2)
        self.write_list = [b"\x40", None]  # Co=0, D/C#=1
        super().__init__(width, height, external_vcc)

    def write_cmd(self, cmd):
        self.temp[0] = 0x80  # Co=1, D/C#=0
        self.temp[1] = cmd
        self.i2c.writeto(self.addr, self.temp)

    def write_data(self, buf):
        self.write_list[1] = buf
        self.i2c.writevto(self.addr, self.write_list)

# ECE 2 - Sampling and aliasing
# The PWM hardware makes a square wave on GPIO 26 (the LED shows it).
# A wire carries it to GPIO 35, which the CPU samples at a fixed rate FS.
from machine import Pin, PWM
import time

F_SIGNAL = 50            # Hz, square wave from the PWM hardware
FS = 1000                # samples per second
N = 1000                 # samples to take (1 s at 1 kHz)

src = PWM(Pin(26), freq=F_SIGNAL, duty_u16=32768)    # 50 % duty
inp = Pin(35, Pin.IN)

samples = bytearray(N)
period_us = 1_000_000 // FS
t = time.ticks_us()
for i in range(N):
    while time.ticks_diff(time.ticks_us(), t) < 0:   # wait for the next sample instant
        pass
    samples[i] = inp.value()
    t = time.ticks_add(t, period_us)

rises = sum(1 for i in range(1, N) if samples[i] and not samples[i - 1])
print("Signal %d Hz, sampled at %d Hz for %.2f s" % (F_SIGNAL, FS, N / FS))
print("Rising edges seen: %d -> apparent frequency about %.0f Hz" % (rises, rises * FS / N))
print("First 80 samples:", "".join("#" if s else "_" for s in samples[:80]))
The PWM hardware runs on its own, like a signal generator. The program only takes a sample every 1/FS seconds and counts the rising edges it saw. If FS is too low, the samples tell a false story: that is aliasing.

adc = ADC(Pin(34))
adc.atten(ADC.ATTN_11DB)

NOISE = 300              # +/- ADC counts of added noise (try 0, 100, 600)
M = 8                    # moving-average length = number of FIR taps
ALPHA = 0.2              # IIR smoothing factor, 0 < ALPHA <= 1

buf = [0] * M
idx = 0
iir = None
while True:
    clean = adc.read_u16() >> 4                          # 0 ... 4095
    x = clean + random.randint(-NOISE, NOISE)            # noisy sample x[n]
    buf[idx] = x
    idx = (idx + 1) % M
    fir = sum(buf) // M                                  # y[n] = (x[n] + ... + x[n-M+1]) / M
    iir = x if iir is None else iir + ALPHA * (x - iir)  # y[n] = y[n-1] + a(x[n] - y[n-1])
    print("clean:%4d noisy:%4d FIR:%4d IIR:%4d" % (clean, x, fir, int(iir)))
    time.sleep_ms(50)                                    # 20 samples per second
FIR (moving average): y[n] = (x[n] + x[n−1] + … + x[n−M+1]) / M. IIR (exponential): y[n] = y[n−1] + α(x[n] − y[n−1]). Both are low-pass filters. Watch the four columns as you turn the knob.

Try this

  1. Hold the knob still. Roughly how much do noisy, FIR and IIR jump about? Moving-average noise should shrink by about √M.
  2. Turn the knob quickly from one end to the other (a step). How many samples does each filter take to settle?
  3. Try M = 2, 8 and 32, and α = 0.05, 0.2 and 0.8. What is the trade-off?
  4. Compare the memory each filter needs, and the work per sample. Which would you put on a tiny battery sensor?
  5. Sampling at 20 Hz, what is the first frequency the M = 8 moving average blocks completely? Deep dive
Answers
  1. The FIR noise is about 1/√8 ≈ 0.35 of the raw noise. The IIR with α = 0.2 is similar.
  2. The FIR settles in exactly M samples. The IIR approaches the new value gradually, reaching about 63 % after roughly 1/α = 5 samples.
  3. More smoothing (larger M, smaller α) means slower response: the classic noise vs delay trade-off.
  4. FIR: M stored samples and M additions. IIR: one stored value and one multiply-add. The IIR is far cheaper, which is why embedded code uses it so often.
  5. The moving average has zeros at multiples of FS/M = 20/8 = 2.5 Hz, so a 2.5 Hz component is removed completely.
